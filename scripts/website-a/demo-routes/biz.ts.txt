import { NextResponse } from "next/server";
import { requireCoach } from "../../../lib/auth";
import { getData, persist } from "../../../lib/db";

// The website's demo (ironline-demo only): the coach's business name set or cleared (?name=), so the client app's top
// bar shows NORCOACH for the store's pictures and the business for the website's.
export const dynamic = "force-dynamic";

export async function GET(request: Request) {
  const coach = await requireCoach();
  const name = new URL(request.url).searchParams.get("name") ?? "";
  const user = getData().users.find((u) => u.id === coach.id)!;
  user.coach_settings = { ...(user.coach_settings ?? {}), business: { ...(user.coach_settings?.business ?? {}), business_name: name || undefined } };
  persist();
  return NextResponse.json({ ok: true, business: name || null });
}
